function passesFilter(item){

  if(activeFilter === 'new' && !item.new) return false;

  if(activeFilter === 'link' && !item.link) return false;

  if(activeFilter === 'key' && !item.key) return false;

  if(priceRange !== 'all'){

    const [lo, hi] = priceRange.split('-');

    if(hi && p >= Number(hi)) return false;

  return true;
